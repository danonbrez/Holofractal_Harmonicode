cmake_minimum_required(VERSION 3.16)
project(hhs_pass159 C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(hhs_pass159 SHARED src/hhs159_core.c ../../hhs_runtime/src/hhs_hash216.c)
target_include_directories(hhs_pass159 PUBLIC include src ../../hhs_runtime/include)
target_compile_options(hhs_pass159 PRIVATE -Wall -Wextra -Werror -pedantic)
add_executable(harmonicode src/hhs159_cli.c)
target_link_libraries(harmonicode PRIVATE hhs_pass159)
add_executable(hhs-pass159-verify tests/test_pass159.c)
target_link_libraries(hhs-pass159-verify PRIVATE hhs_pass159)
enable_testing()
add_test(NAME pass159-foundation COMMAND hhs-pass159-verify)
