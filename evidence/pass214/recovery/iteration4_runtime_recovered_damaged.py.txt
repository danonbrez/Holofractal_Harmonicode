def _apply_adapter(value: Any, adapter: Mapping[str, Any]) -> Any:b    kind = adapter.get("class", "IDENTITY")
    config = adapter.get("config", {})
    if kind not in ALLOWED_ADAPTERS or not isinstance(config, Mapping):b        raise Pass214CallableOracleError("PASS214_ITERATION4_ADAPTER_INVALID")b    if kind == "IDENTITY":b        return value
    if kind == "CANONICAL_BYTES_ENVELOPE":b        return {"canonical_utf8_hex": canonical_json(value).encode("utf-8").hex()}b    if not isinstance(value, Mapping):b        raise Pass214CallableOracleError("PASS214_ITERATION4_MAPPING_ADAPTER_REQUIRES_MAPPING")b    if kind == "KEY_RENAME":b        mapping = config.get("mapping")b        if not isinstance(mapping, Mapping) or not all(isinstance(k, str) and isinstance(v, str) for k, v in mapping.items()):b            raise Pass214CallableOracleError("PASS214_ITERATION4_KEY_RENAME_CONFIG_INVALID")b        if len(set(mapping.values())) != len(mapping):b            raise Pass214CallableOracleError("PASS214_ITERATION4_KEY_RENAME_COLLISION")
        return {mapping.get(k, k): value[k] for k in sorted(value)}b    fields = config.get("fields")
    if not isinstance(fields, list) or not all(isinstance(item, str) for item in fields) or len(fields) != len(set(fields)):b        raise Pass214CallableOracleError("PASS214_ITERATION4_FIELD_PROJECTION_CONFIG_INVALID")b    if any(item not in value for item in fields):b        raise Pass214CallableOracleError("PASS214_ITERATION4_FIELD_PROJECTION_MISSING_FIELD")b    return {item: value[item] for item in fields}
